SQL

table

  1. select * from table_name;

    • select column_name / top / percent / distinct … as column_alias
      • aggregate functions: count() / sum() / avg() / min() / max()
    • order by column_name asc/desc;
    • group by column_name;
      • having aggregate_function(column_name) condition;
    • limit n;
    • where column_name = val;
      • and / or / not
      • is null / is not null
      • in / not in ()
      • between val1 and val2
      • like ‘%val%’
      • join
        • left join / right join / inner join / full join on table1.column_name = table2.column_name
          • Returns all rows from the left/right table, and only the matched rows from the right/left table
      • union / union all
        • combine the result set of two or more select statements
        • union removes duplicate rows, while union all includes duplicates
        • must have an equal number of expressions in their target lists.
      • exists(subquery)
      • column_name = any / all(subquery)
    • case
    • SELECT * INTO newtable [IN external_db] from …
      • to create a new table and fill it with data from an existing table.
      • useful for creating backups or for creating a temporary table for analysis.
    • INSERT INTO target_table SELECT * FROM source_table WHERE condition;
      • to copy data from an existing table and insert it into another existing table.
      • the data types in source and target tables match.

  2. insert into table_name (col1, col2, …) values (val1, val2, …);

  3. update table_name set col1 = val1, col2 = val2, …

  4. delete from table_name

    • where col1 = val1

  5. create table table_name(col1 datatype PRIMARY KEY, col2 datatype NOT NULL, …)

    • datatype: int, char, var(255),
    • constraints: not null, unique, primary key, foreign key, check(condition), default
    • AUTO_INCREMENT
    • CREATE [UNIQUE] INDEX index_name on table_name(col1, col2, …)

  6. drop table table_name

  7. alert table table_name

    • add col_name datatype
    • DROP COLUMN column_name;
    • RENAME COLUMN old_name to new_name;
    • RENAME TO new_table_name;
    • ALTER COLUMN column_name new_datatype constraint;
    • ADD CONSTRAINT constraint_name constraint_definition;

  8. views
